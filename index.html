import java.net.URI; import java.net.http.HttpClient; import java.net.http.HttpRequest; import java.net.http.HttpResponse; public class GeminiService { private static final String API_KEY = "YOUR_API_KEY_HERE"; private static final String URL = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=" + API_KEY; public String askGemini(String userPrompt) throws Exception { // Constructing the JSON body for "responds to everything" String jsonBody = """ { "contents": [{ "parts":[{"text": "%s"}] }] } """.formatted(userPrompt); HttpClient client = HttpClient.newHttpClient(); HttpRequest request = HttpRequest.newBuilder() .uri(URI.create(URL)) .header("Content-Type", "application/json") .POST(HttpRequest.BodyPublishers.ofString(jsonBody)) .build(); HttpResponse response = client.send(request, HttpResponse.BodyHandlers.ofString()); return response.body(); } }
